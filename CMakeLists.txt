cmake_minimum_required(VERSION 3.25)
project(offcut VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8 /W4 /EHsc)
  add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
endif()
option(OFFCUT_BUILD_APP "Build the native desktop app" ON)
# The prompt files (docs/ux-design.md section 8) are read at run time from
# data/prompts; a copy of each is compiled in as the fallback.
set(OFFCUT_PROMPT_FILES brief.txt reimagine.txt refine.txt fix-support.txt describe-scene.txt describe-element.txt preferences.txt output.txt
    review-chips.txt director-plan.txt director-critique.txt)
set(builtin_prompts "// Generated by CMake from data/prompts: the compiled-in copies.\n#include <string_view>\nnamespace offcut\n{\nstd::string_view builtin_prompt_text(std::string_view name)\n{\n")
foreach(f ${OFFCUT_PROMPT_FILES})
  file(READ ${CMAKE_CURRENT_SOURCE_DIR}/data/prompts/${f} content)
  string(APPEND builtin_prompts "    if (name == \"${f}\")\n        return R\"offcut_prompt(${content})offcut_prompt\";\n")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/data/prompts/${f})
endforeach()
string(APPEND builtin_prompts "    return {};\n}\n}  // namespace offcut\n")
file(CONFIGURE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/builtin_prompts.cpp CONTENT "${builtin_prompts}" @ONLY)
add_library(offcut_core src/project.cpp src/transform.cpp src/history.cpp src/guidance.cpp src/export.cpp src/ai_client.cpp src/ai_provider.cpp src/chatgpt_auth.cpp src/chatgpt_store.cpp src/ai_responses.cpp src/director.cpp src/local_ops.cpp
            src/prompts.cpp src/versions.cpp src/ai_workflow.cpp src/auto_loop.cpp ${CMAKE_CURRENT_BINARY_DIR}/builtin_prompts.cpp)
if(WIN32)
  target_link_libraries(offcut_core PUBLIC winhttp advapi32 bcrypt cabinet ws2_32 shell32 ole32)
endif()
target_include_directories(offcut_core PUBLIC src)
# A development build reads the source tree's prompt files, so wording is
# tuned without rebuilding.
target_compile_definitions(offcut_core PRIVATE OFFCUT_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
if(OFFCUT_BUILD_APP)
  set(RAPPORT_STATIC ON CACHE BOOL "" FORCE)
  set(RAPPORT_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(RAPPORT_TESTS OFF CACHE BOOL "" FORCE)
  set(RAPPORT_SHELL OFF CACHE BOOL "" FORCE)
  set(RAPPORT_BACKEND_GTK4 OFF CACHE BOOL "" FORCE)
  add_subdirectory(rapport)
  add_executable(offcut src/main.cpp src/director_cli.cpp src/app.cpp src/window.cpp src/image_io.cpp src/settings.cpp src/ai_providers_page.cpp src/smoke.cpp src/ai_request.cpp src/window_parts.cpp src/window_canvas.cpp src/window_ai.cpp)
  target_link_libraries(offcut PRIVATE offcut_core rapport::rapport)
  # The prompt files beside the exe, where an installed Offcut reads them.
  add_custom_command(TARGET offcut POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_CURRENT_SOURCE_DIR}/data/prompts
                                                      $<TARGET_FILE_DIR:offcut>/data/prompts)
  set(builtins "#include <rapport/rapport.h>\n")
  foreach(backend win32 gtk3 qt6 null)
    if(TARGET rapport::backend_${backend})
      target_link_libraries(offcut PRIVATE rapport::backend_${backend})
      string(APPEND builtins "RAPPORT_BUILTIN_BACKEND(${backend});\n")
    endif()
  endforeach()
  file(GENERATE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/builtin_backends.cpp CONTENT "${builtins}")
  target_sources(offcut PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/builtin_backends.cpp)
  if(WIN32)
    enable_language(RC)
    target_sources(offcut PRIVATE data/windows/offcut.rc)
    set_target_properties(offcut PROPERTIES WIN32_EXECUTABLE ON)
    if(MSVC)
      target_link_options(offcut PRIVATE /ENTRY:mainCRTStartup /MANIFEST:NO)
    endif()
  endif()
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(offcut-tests tests/core_test.cpp tests/director_test.cpp tests/local_ops_test.cpp)
  target_link_libraries(offcut-tests PRIVATE offcut_core)
  add_test(NAME offcut-core COMMAND offcut-tests)
endif()
