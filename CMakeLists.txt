cmake_minimum_required(VERSION 3.25)
project(offcut VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8 /W4 /EHsc)
  add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
endif()
option(OFFCUT_BUILD_APP "Build the native desktop app" ON)
add_library(offcut_core src/project.cpp src/transform.cpp src/export.cpp)
target_include_directories(offcut_core PUBLIC src)
if(OFFCUT_BUILD_APP)
  set(RAPPORT_STATIC ON CACHE BOOL "" FORCE)
  set(RAPPORT_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(RAPPORT_TESTS OFF CACHE BOOL "" FORCE)
  set(RAPPORT_SHELL OFF CACHE BOOL "" FORCE)
  set(RAPPORT_BACKEND_GTK4 OFF CACHE BOOL "" FORCE)
  add_subdirectory(rapport)
  add_executable(offcut src/main.cpp src/app.cpp src/window.cpp src/image_io.cpp src/settings.cpp src/smoke.cpp)
  target_link_libraries(offcut PRIVATE offcut_core rapport::rapport)
  set(builtins "#include <rapport/rapport.h>\n")
  foreach(backend win32 gtk3 qt6 null)
    if(TARGET rapport::backend_${backend})
      target_link_libraries(offcut PRIVATE rapport::backend_${backend})
      string(APPEND builtins "RAPPORT_BUILTIN_BACKEND(${backend});\n")
    endif()
  endforeach()
  file(GENERATE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/builtin_backends.cpp CONTENT "${builtins}")
  target_sources(offcut PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/builtin_backends.cpp)
  if(WIN32)
    enable_language(RC)
    target_sources(offcut PRIVATE data/windows/offcut.rc)
    set_target_properties(offcut PROPERTIES WIN32_EXECUTABLE ON)
    if(MSVC)
      target_link_options(offcut PRIVATE /ENTRY:mainCRTStartup /MANIFEST:NO)
    endif()
  endif()
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(offcut-tests tests/core_test.cpp)
  target_link_libraries(offcut-tests PRIVATE offcut_core)
  add_test(NAME offcut-core COMMAND offcut-tests)
endif()
